  'Rivers', 'Sokoto', 'Taraba', 'Yobe', 'Zamfara', 'Abuja (FCT)', 'Others'
];

function CompactSubscribeForm({ 
  title, 
  subtitle, 
  disclaimer, 
  buttonText 
}: { 
  title: string; 
  subtitle: string; 
  disclaimer: string; 
  buttonText: string;
  buttonContext?: string;
}) {
  const [step, setStep] = useState(1);
  const [email, setEmail] = useState('');
  const [firstName, setFirstName] = useState('');
  const [surname, setSurname] = useState('');
  const [occupation, setOccupation] = useState('');
  const [stateOfRes, setStateOfRes] = useState('');
  const [customState, setCustomState] = useState('');
  const [consent, setConsent] = useState(true);
  
  const [token, setToken] = useState('');
  const [submitting, setSubmitting] = useState(false);
  const [error, setError] = useState('');

  const handleStep1 = async (e: React.FormEvent) => {
    e.preventDefault();
    const emailLower = email.toLowerCase().trim();
    if (!emailLower || !emailLower.includes('@')) {
      setError('Please enter a valid email address.');
      return;
    }

    setSubmitting(true);
    setError('');

    try {
      const res = await fetch('/api/subscribe', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({ email: emailLower }),
      });
      const data = await res.json();

      if (data.success) {
        if (data.status === 'already_subscribed') {
          setError(data.message || 'This email address is already subscribed.');
        } else {
          if (data.token) {
            setToken(data.token);
          }
          setStep(2);
        }
      } else {
        setError(data.error || 'Failed to start subscription. Please try again.');
      }
    } catch (err: any) {
      console.error('Subscribe API error:', err);
      setError(err.message || 'Network error starting subscription.');
    } finally {
      setSubmitting(false);
    }
  };

  const handleStep2 = async (e: React.FormEvent) => {
    e.preventDefault();
    if (!firstName.trim() || !surname.trim()) {
      setError('Please enter your First Name and Surname.');
      return;
    }
    if (!occupation) {
      setError('Please select your Occupation.');
      return;
    }
    if (!stateOfRes) {
      setError('Please select your State of residence.');
      return;
    }
    if (stateOfRes === 'Others' && !customState.trim()) {
      setError('Please specify your state of residence or country.');
      return;
    }
    if (!consent) {
      setError('You must accept the terms to complete subscription.');
      return;
    }

    setSubmitting(true);
    setError('');

    try {
      const res = await fetch('/api/complete-subscription', {
        method: 'POST',
        headers: { 'Content-Type': 'application/json' },
        body: JSON.stringify({
          token,
          firstName: firstName.trim(),
          surname: surname.trim(),
          occupation,
          stateOfOrigin: stateOfRes === 'Others' ? customState.trim() : stateOfRes,
        }),
      });
      const data = await res.json();

      if (data.success) {
        setStep(3);
      } else {
        setError(data.error || 'Failed to complete subscription. Please try again.');
      }
    } catch (err: any) {
      console.error('Complete Subscribe API error:', err);
      setError(err.message || 'Network error completing subscription.');
    } finally {
      setSubmitting(false);
    }
  };

  if (step === 3) {
    return (
      <div className="mt-6 mb-6 p-8 bg-surface-container-low border border-outline-variant rounded-lg font-sans text-center">
        <div className="w-12 h-12 rounded-full bg-emerald-100 text-emerald-600 flex items-center justify-center mx-auto mb-4">
          <Check className="w-6 h-6 stroke-[3]" />
        </div>
        <h3 className="text-xl font-bold text-primary mb-2">Subscription Complete!</h3>
        <p className="text-sm text-on-surface-variant leading-relaxed">
          Thank you for joining. You will now receive our latest updates and analysis.
        </p>
      </div>
    );
  }

  return (
    <div className="mt-6 mb-6 p-6 bg-surface-container-low border border-outline-variant rounded-lg font-sans text-left">
      {step === 1 && (
        <form onSubmit={handleStep1}>
          <h3 className="text-lg font-bold text-primary mb-2">{title}</h3>
          <p className="text-sm text-on-surface-variant mb-4 leading-relaxed">{subtitle}</p>
          
          <div className="flex flex-col sm:flex-row gap-3">
            <input 
              type="email" 
              value={email}
              onChange={(e) => setEmail(e.target.value)}
              placeholder="Enter email address" 
              required
              className="flex-grow px-4 py-3 border border-outline-variant rounded focus:outline-none focus:ring-1 focus:ring-primary focus:border-primary text-sm bg-white"
            />
            <button 
              type="submit" 
              disabled={submitting}
              className="whitespace-nowrap px-6 py-3 bg-primary text-white rounded font-bold text-sm hover:bg-primary/90 transition-colors shadow-sm disabled:opacity-50"
            >
              {submitting ? 'Sending...' : buttonText}
            </button>
          </div>
          {error && <p className="text-xs text-red-600 font-semibold mt-2">{error}</p>}
          <p className="text-xs text-outline mt-3 font-medium">{disclaimer}</p>
        </form>
      )}

      {step === 2 && (
        <form onSubmit={handleStep2} className="animate-fade-in text-left">
          <h3 className="text-lg font-bold text-primary mb-2">Complete Your Details</h3>
          <p className="text-sm text-on-surface-variant mb-4 leading-relaxed">Just a few more details to set up your subscription.</p>
          
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
            <div>
              <label className="block text-xs font-bold text-primary mb-1">First name *</label>
              <input
                type="text"
                value={firstName}
                onChange={(e) => setFirstName(e.target.value)}
                required
                className="w-full px-3 py-2 border border-outline-variant rounded text-sm bg-white focus:outline-none focus:border-primary"
              />
            </div>
            <div>
              <label className="block text-xs font-bold text-primary mb-1">Surname *</label>
              <input
                type="text"
                value={surname}
                onChange={(e) => setSurname(e.target.value)}
                required
                className="w-full px-3 py-2 border border-outline-variant rounded text-sm bg-white focus:outline-none focus:border-primary"
              />
            </div>
          </div>
          
          <div className="grid grid-cols-1 sm:grid-cols-2 gap-4 mb-4">
            <div>
              <label className="block text-xs font-bold text-primary mb-1">Occupation *</label>
              <select
                value={occupation}
                onChange={(e) => setOccupation(e.target.value)}
                required
                className="w-full px-3 py-2 border border-outline-variant rounded text-sm bg-white focus:outline-none focus:border-primary"
              >
                <option value="" disabled>Select occupation</option>
                {OCCUPATIONS.map(occ => <option key={occ} value={occ}>{occ}</option>)}
              </select>
            </div>
            <div>
              <label className="block text-xs font-bold text-primary mb-1">State of Residence *</label>
              <select
                value={stateOfRes}
                onChange={(e) => setStateOfRes(e.target.value)}
                required
                className="w-full px-3 py-2 border border-outline-variant rounded text-sm bg-white focus:outline-none focus:border-primary"
              >
                <option value="" disabled>Select state</option>
                {STATES.map(s => <option key={s} value={s}>{s}</option>)}
              </select>
            </div>
          </div>
          
          {stateOfRes === 'Others' && (
            <div className="mb-4">
              <label className="block text-xs font-bold text-primary mb-1">Specify State or Country *</label>
              <input
                type="text"
                value={customState}
                onChange={(e) => setCustomState(e.target.value)}
                required
                className="w-full px-3 py-2 border border-outline-variant rounded text-sm bg-white focus:outline-none focus:border-primary"
              />
            </div>
          )}

          <div className="mb-4 flex items-start gap-2">
            <input
              type="checkbox"
              id="consent-inline"
              checked={consent}
              onChange={(e) => setConsent(e.target.checked)}
              className="mt-1"
